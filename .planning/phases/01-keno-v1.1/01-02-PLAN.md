```python
def render_sidebar(n_draws: int) -> Dict:
    """Render sidebar controls and return configuration dict.
    
    Per D-10: No preset selector — always Personalizada.
    Per D-08: Band values auto-recalculate when pool_size changes.
    Per D-09: Block generation if Baja+Media+Alta != pool_size.
    Per D-11: Band metrics displayed with colored text.
    """
    st.sidebar.header(":material/tune: Configuracion")

    # --- Window slider (CTRL-01) ---
    max_window = min(100, n_draws)
    window = st.sidebar.slider(
        "Ventana de sorteos (retroactivos)",
        min_value=10,
        max_value=max_window,
        value=max_window,
        help="Maximo 100 sorteos retroactivos desde el ultimo sorteo cargado.",
    )

    st.sidebar.divider()

    # --- Pool size (CTRL-02) ---
    pool_size = st.sidebar.slider(
        "Tamano del Pool Dinamico",
        min_value=15,
        max_value=30,
        value=20,
        help="Cantidad de numeros en el pool generado automaticamente.",
    )

    # --- Ticket quantity (CTRL-03) ---
    n_tickets = st.sidebar.slider(
        "Cantidad de Boletos (Wheeling)",
        min_value=6,
        max_value=30,
        value=18,
        help="Numero de combinaciones a generar via reduccion combinatoria.",
    )

    st.sidebar.divider()

    # --- Band Distribution (CTRL-04, D-10: forced Personalizada) ---
    st.sidebar.subheader("Distribucion por Franja")
    st.sidebar.caption("Baja: 01-26 | Media: 27-54 | Alta: 55-80")

    # D-08: Auto-recalc proportionally when pool_size changes
    # Use session_state to track previous pool_size for recalc
    prev_pool = st.session_state.get("_prev_pool_size", pool_size)
    if pool_size != prev_pool and "_band_baja" in st.session_state:
        # Recalculate proportionally
        old_baja = st.session_state["_band_baja"]
        old_media = st.session_state["_band_media"]
        old_alta = st.session_state["_band_alta"]
        old_total = old_baja + old_media + old_alta
        if old_total > 0:
            new_baja = max(0, round(pool_size * old_baja / old_total))
            new_media = max(0, round(pool_size * old_media / old_total))
            new_alta = pool_size - new_baja - new_media  # remainder goes to Alta
            # Ensure non-negative
            if new_alta < 0:
                new_alta = 0
                new_media = pool_size - new_baja
            st.session_state["_band_baja"] = new_baja
            st.session_state["_band_media"] = new_media
            st.session_state["_band_alta"] = new_alta
    st.session_state["_prev_pool_size"] = pool_size

    # Default proportional values for first load
    default_baja = st.session_state.get("_band_baja", max(1, round(pool_size * 26 / 80)))
    default_media = st.session_state.get("_band_media", max(1, round(pool_size * 28 / 80)))
    default_alta = st.session_state.get("_band_alta", pool_size - default_baja - default_media)
    # Ensure defaults are within bounds
    default_baja = max(0, min(default_baja, pool_size))
    default_media = max(0, min(default_media, pool_size - default_baja))
    default_alta = pool_size - default_baja - default_media

    # Three number inputs per D-10 (forced Personalizada)
    col1, col2, col3 = st.sidebar.columns(3)
    with col1:
        baja = st.number_input(
            "Baja",
            min_value=0,
            max_value=pool_size,
            value=default_baja,
            key="_band_baja",
            help="Numeros en rango 01-26",
        )
    with col2:
        media = st.number_input(
            "Media",
            min_value=0,
            max_value=pool_size,
            value=default_media,
            key="_band_media",
            help="Numeros en rango 27-54",
        )
    with col3:
        alta = st.number_input(
            "Alta",
            min_value=0,
            max_value=pool_size,
            value=default_alta,
            key="_band_alta",
            help="Numeros en rango 55-80",
        )

    total = baja + media + alta

    # D-11: Colored band metrics
    st.sidebar.divider()
    col_b, col_m, col_a = st.sidebar.columns(3)
    with col_b:
        st.markdown(f"<div style='text-align:center'><span style='color:#0969DA;font-weight:bold;font-size:1.2em'>{baja}</span><br><span style='font-size:0.8em'>Baja</span></div>", unsafe_allow_html=True)
    with col_m:
        st.markdown(f"<div style='text-align:center'><span style='color:#BF8700;font-weight:bold;font-size:1.2em'>{media}</span><br><span style='font-size:0.8em'>Media</span></div>", unsafe_allow_html=True)
    with col_a:
        st.markdown(f"<div style='text-align:center'><span style='color:#CF222E;font-weight:bold;font-size:1.2em'>{alta}</span><br><span style='font-size:0.8em'>Alta</span></div>", unsafe_allow_html=True)

    # D-09: Sum validation — BLOCK if mismatch
    band_dist = (baja, media, alta)
    if total != pool_size:
        st.sidebar.error(
            f":material/error: La suma ({total}) no coincide con el pool ({pool_size}). "
            f"Ajuste los valores."
        )
        # Signal to caller that band_dist is invalid
        st.session_state["_band_valid"] = False
    else:
        st.session_state["_band_valid"] = True

    return {
        "window": window,
        "pool_size": pool_size,
        "n_tickets": n_tickets,
        "band_dist": band_dist,
        "band_valid": st.session_state.get("_band_valid", True),
    }
```
